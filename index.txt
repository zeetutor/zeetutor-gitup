function prettyName(path) {
  return String(path || '').replace(/\/index\.html$/i, '').replace(/[-_]+/g, ' ').replace(/\s+/g, ' ').trim().replace(/\b\w/g, c => c.toUpperCase());
}
function formatTime(ms) {
  const total = Math.max(0, Math.ceil(ms / 1000));
  return String(Math.floor(total / 60)).padStart(2, '0') + ':' + String(total % 60).padStart(2, '0');
}
function pad(n) { return String(n).padStart(2, '0'); }
function todayKey() { const d = new Date(); return d.getFullYear() + '-' + pad(d.getMonth() + 1) + '-' + pad(d.getDate()); }
function nowTime() { const d = new Date(); return pad(d.getHours()) + ':' + pad(d.getMinutes()) + ':' + pad(d.getSeconds()); }
function niceDate() { return new Date().toLocaleDateString(undefined, { weekday: 'long', day: 'numeric', month: 'short', year: 'numeric' }); }
function buildWorksheetUrl(path) {

  if (!rawUrl.toLowerCase().endsWith('.html')) rawUrl += '/index.html';
  return rawUrl.split('/').map(p => encodeURIComponent(p)).join('/') + '#room=' + roomId;
}
function remainingMs() { return sessionStart ? durationMs + extraMs - (Date.now() - sessionStart) : null; }
function extraMinutes() { return extraAdded ? Math.round(extraMs / 60000) : 0; }
function classMinutes() { return (Math.round(durationMs / 60000)) + extraMinutes(); }
function clampMinutes(n) {
  n = parseInt(n, 10);
  if (!n || n < 1) n = 1;
  if (n > 30) n = 30;
  return n;
}

window.markAttend = function() {
  if (!studentName) return showToast('Wait for teacher to choose student');
  const date = todayKey();
  const prev = attendanceData.days?.[date]?.[studentName] || {};
  const record = Object.assign({}, prev, {
    name: studentName,
    attended: true,
    date,
    attendTime: nowTime(),
    endTime: prev.endTime || '',
    baseMinutes: 30,
    extraMinutes: extraMinutes(),
    totalMinutes: classMinutes(),
    review: prev.review || ''
  });
  studentAttendedToday = true;
  if (!attendanceData.days) attendanceData.days = {};
  if (!attendanceData.days[date]) attendanceData.days[date] = {};
  attendanceData.days[date][studentName] = record;
  saveJson();
  hideAttend();
  showToast(studentName + ' attended');
};

function updateAttendanceDuration() {
  if (!studentName) return;
  const date = todayKey();
  const rec = attendanceData.days?.[date]?.[studentName];
  if (!rec || !rec.attended) return;
  rec.extraMinutes = extraMinutes();
  rec.totalMinutes = classMinutes();
  rec.endTime = nowTime();
  saveJson();
}

window.deleteDayEntry = function(date, name) {
  if (!confirm('Erase ' + name + ' on ' + date + ' from JSON?')) return;

window.saveReview = function(date, name, value) {
  if (!attendanceData.days) attendanceData.days = {};

  attendanceData.days[date][name].review = value;

};

window.importAttendanceJson = function(event) {
  const file = event.target.files && event.target.files[0];
  event.target.value = '';
  if (!file) return;
  const reader = new FileReader();
  reader.onload = () => {
    try {
      const parsed = JSON.parse(reader.result);
      if (!parsed || typeof parsed !== 'object') throw new Error('bad');
      attendanceData = {
        students: Array.isArray(parsed.students) ? parsed.students : studentNames.slice(),
        days: parsed.days && typeof parsed.days === 'object' ? parsed.days : {}
      };
      if (attendanceData.students.length) {
        studentNames = attendanceData.students.map(n => String(n).toUpperCase());
        if (studentsRef) set(studentsRef, studentNames);
      }
      saveJson();
      renderAttendanceReview();
      showToast('JSON imported. Previous live JSON replaced.');
    } catch (e) {
      showToast('Invalid JSON file');
    }
  };
  reader.readAsText(file);
};

function renderAttendanceReview() {
  const box = document.getElementById('attendance-review');
  const days = Object.keys(attendanceData.days || {}).sort().reverse();
  if (!days.length) {
    box.innerHTML = '<p class="py-16 text-center font-bold text-slate-400">JSON is empty. Student must press Attend.</p>';
    return;
  }
  box.innerHTML = days.map(date => {
    const row = attendanceData.days[date] || {};
    const names = Array.from(new Set([...(studentNames || []), ...Object.keys(row)]));
    const cards = names.map(name => {
      const rec = row[name] || {};
      const safeDate = esc(date);
      const safeName = esc(name);
      const review = (rec.review || '').replace(/</g, '');
      if (!rec.attended) {
        return `<div class="rounded-2xl glass-panel p-4">
          <div class="flex justify-between"><p class="font-fredoka text-2xl">${name}</p>
          <button onclick="deleteDayEntry('${safeDate}','${safeName}')" class="text-xs bg-rose-900 px-2 py-1 rounded-lg font-bold">Delete</button></div>
          <p class="text-slate-400 font-bold">Not attended</p>
          <textarea onchange="saveReview('${safeDate}','${safeName}', this.value)" placeholder="Review of ${name}" class="mt-2 w-full bg-slate-950/80 border border-slate-700 rounded-xl p-2 text-sm text-white focus:outline-none">${review}</textarea>
        </div>`;
      }
      return `<div class="rounded-2xl border border-emerald-500/30 glass-panel p-4">
        <div class="flex justify-between items-start gap-2">
          <p class="font-fredoka text-2xl text-white">${name}</p>
          <button onclick="deleteDayEntry('${safeDate}','${safeName}')" class="text-xs bg-rose-700 px-2 py-1 rounded-lg font-bold">Delete entry</button>
        </div>
        <p class="text-emerald-400 font-bold">Attended ${rec.attendTime || ''}</p>
        <p class="text-amber-300 font-black text-xl mt-1">${rec.totalMinutes || 30} min</p>
        <p class="text-xs text-slate-300 font-bold">30 min class${rec.extraMinutes ? ' + ' + rec.extraMinutes + ' extra' : ''}</p>
        <textarea onchange="saveReview('${safeDate}','${safeName}', this.value)" placeholder="Review of ${name}" class="mt-2 w-full bg-slate-950/80 border border-slate-700 rounded-xl p-2 text-sm text-white focus:outline-none">${review}</textarea>
      </div>`;
    }).join('');
    return `<section class="rounded-3xl border border-white/10 glass-panel p-5">
      <div class="flex justify-between items-center mb-3">
        <h2 class="font-fredoka text-2xl text-amber-300">${date}</h2>
        <button onclick="deleteWholeDay('${esc(date)}')" class="text-xs bg-rose-900 hover:bg-rose-800 px-3 py-1 rounded-lg font-bold">Delete day</button>
      </div>
      <div class="grid grid-cols-1 sm:grid-cols-2 gap-3">${cards}</div>
    </section>`;
  }).join('');
}

function connectFirebaseEngine() {
  dbRef = ref(db, 'global_rooms/' + roomId);
  globalPresetsRef = ref(db, 'shared_presets/list');
  studentsRef = ref(db, 'shared_presets/students');
  attendanceRef = ref(db, 'attendance_log');

  onValue(globalPresetsRef, snap => {
    globalPresets = snap.val() || [];
    if (myRole === 'teacher') renderFileList();
  });
  onValue(studentsRef, snap => {
    const list = snap.val();
    studentNames = Array.isArray(list) && list.length ? list : DEFAULT_STUDENTS.slice();
    if (!document.getElementById('student-select-screen').classList.contains('hidden')) renderEntryNames();
  });
  onValue(attendanceRef, snap => {
    const val = snap.val();
    attendanceData = val && typeof val === 'object' ? val : emptyJson();
    if (!attendanceData.days) attendanceData.days = {};
    if (Array.isArray(attendanceData.students) && attendanceData.students.length) {
      studentNames = attendanceData.students;
    }
    const rec = attendanceData.days[todayKey()]?.[studentName];
    studentAttendedToday = !!(rec && rec.attended);
    if (myRole === 'student') maybeShowAttend();
    if (!document.getElementById('attendance-screen').classList.contains('hidden')) renderAttendanceReview();
  });

  onValue(dbRef, snapshot => {
    const data = snapshot.val();
    if (data) {
      studentName = data.student_name || studentName;
      selectedStudent = selectedStudent || studentName;
      document.getElementById('over-student').innerText = studentName;
      document.getElementById('active-student-name').innerText = studentName ? '· ' + studentName : '';
      document.getElementById('lib-student').innerText = studentName || '';
      const rec = attendanceData.days?.[todayKey()]?.[studentName];
      studentAttendedToday = !!(rec && rec.attended);

      if (data.duration_ms) durationMs = Number(data.duration_ms) || DEFAULT_DURATION_MS;
      if (data.session_start) {
        sessionStart = data.session_start;
        extraMs = data.extra_ms || 0;
        extraAdded = extraMs > 0;
        const left = durationMs + extraMs - (Date.now() - sessionStart);
        if (left > 0) {
          studentSawLiveTimer = true;
          hideSmile();
          startLocalClock();
        } else if (studentSawLiveTimer) {
          startLocalClock();
        } else {
          sessionStart = null;
          extraMs = 0;
          extraAdded = false;
          hideSmile();
          resetClocks();
          if (myRole === 'teacher') update(dbRef, { session_start: null, extra_ms: 0 });
        }
      } else {
        sessionStart = null; extraMs = 0; extraAdded = false; waitingForExtend = false; studentSawLiveTimer = false;
        hideSmile();
        resetClocks();
      }
      if (myRole === 'student') applyStudentScreens();
      if (data.current_url) {
        activeFile = data.current_url;
        hideOverlay('waiting-overlay');
      }
      const runId = data.run_id || data.current_url || '';
      if (data.current_url && runId !== lastLoadedUrl) {
        lastLoadedUrl = runId;
        loadWorksheetFast(data.current_url);
      }
      if (data.last_action && data.last_action.initiator !== myRole) {
        if (data.last_action.actionId && data.last_action.actionId !== lastProcessedOuterActionId) {
          lastProcessedOuterActionId = data.last_action.actionId;
          const frame = document.getElementById('game-frame');
          if (frame && frame.contentWindow) frame.contentWindow.postMessage({ type: 'EXECUTE_UNIVERSAL_ACTION', action: data.last_action }, '*');
          const rpl = document.getElementById('touch-ripple');
          if (rpl && data.pointer) {
            rpl.style.left = data.pointer.x + '%'; rpl.style.top = data.pointer.y + '%';
            rpl.style.display = 'block'; rpl.classList.remove('ripple-active'); void rpl.offsetWidth; rpl.classList.add('ripple-active');
            clearTimeout(window.hideRpl);
            window.hideRpl = setTimeout(() => { rpl.style.display = 'none'; rpl.classList.remove('ripple-active'); }, 1000);
          }
        }
      }
      if (data.pointer && data.pointer.initiator !== myRole) {
        const ptr = document.getElementById('remote-pointer');
        ptr.style.left = data.pointer.x + '%'; ptr.style.top = data.pointer.y + '%';
        ptr.classList.remove('hidden');
        clearTimeout(window.hidePtr);
        window.hidePtr = setTimeout(() => ptr.classList.add('hidden'), 2500);
      }
    } else if (myRole === 'teacher') {
      set(dbRef, { current_url: '', last_action: null, extra_ms: 0, student_name: '', session_start: null, duration_ms: null });
    }
  });
}

function resetClocks() {
  ['header-clock', 'lib-clock', 'student-clock-text'].forEach(id => {
    const el = document.getElementById(id);
    if (el) el.textContent = '--:--';
  });
}

function startLocalClock() {
  if (clockInterval) return;
  tickClock();
  clockInterval = setInterval(tickClock, 250);
}

function paintClock(el, remaining) {
  if (!el) return;
  el.textContent = formatTime(remaining);
  el.classList.remove('clock-danger');
  if (remaining <= 60000) el.classList.add('clock-danger');
}

function tickClock() {
  if (!sessionStart) return;
  const left = remainingMs();
  const elapsed = Date.now() - sessionStart;
  const clocks = [
    document.getElementById('header-clock'),
    document.getElementById('lib-clock'),
    document.getElementById('student-clock-text')
  ];

  if (!extraAdded && elapsed >= durationMs) {
    clocks.forEach(c => paintClock(c, 0));
    updateAttendanceDuration();
    if (myRole === 'student' && !studentSawLiveTimer) {
      hideSmile();
      showAttend();
      return;
    }
    if (!waitingForExtend) showSmile(false);
    return;
  }
  if (left > 0) studentSawLiveTimer = true;
  clocks.forEach(c => paintClock(c, left));
  if (extraAdded && smileShown && left > 0) hideSmile();
  if (extraAdded && left <= 0) {
    clocks.forEach(c => paintClock(c, 0));
    updateAttendanceDuration();
    showSmile(true);
    setTimeout(teacherLogout, 1500);
  }
}

function showSmile(finalEnd) {
  smileShown = true;
  waitingForExtend = !finalEnd;
  hideAttend();
  if (myRole === 'teacher') showTeacherCanvas();
  document.getElementById('over-student').innerText = studentName;
  showOverlay('smile-overlay');
  const sub = document.getElementById('smile-sub');
  if (finalEnd || myRole !== 'teacher') {
    hideOverlay('extra-time-controls');
    sub.innerText = finalEnd ? (extraAdded ? 'Extra time finished. Logging out...' : 'Session completed. Logging out...') : '';
    return;
  }
  showOverlay('extra-time-controls');
  sub.innerText = '';
}

function hideSmile() {
  smileShown = false;
  waitingForExtend = false;
  hideOverlay('smile-overlay');
  hideOverlay('extra-time-controls');
}

window.addExtraMinutes = function(mins) {
  if (myRole !== 'teacher' || !dbRef || !sessionStart) return;
  extraAdded = true; extraMs = mins * 60 * 1000; waitingForExtend = false;
  update(dbRef, { extra_ms: extraMs });
  updateAttendanceDuration();
  hideSmile();
  showToast('Extra +' + mins + ' minutes');
};

window.teacherLogout = function() {
  if (loggingOut) return;
  loggingOut = true;
  updateAttendanceDuration();
  if (clockInterval) { clearInterval(clockInterval); clockInterval = null; }
  showToast('Session ended. Logging out...');
  const finish = () => setTimeout(() => location.reload(), 700);
  if (dbRef) set(dbRef, { current_url: '', last_action: null, extra_ms: 0, student_name: '', session_start: null, duration_ms: null }).then(finish).catch(finish);
  else finish();
};

window.addEventListener('message', (event) => {
  if (!dbRef) return;
  if (event.data?.type === 'IFRAME_ACTION_TRIGGERED') {
    const payload = event.data.payload;
    if (payload?.type === 'UNIVERSAL_CURSOR') {
      const now = Date.now();
      if (now - lastFirebaseCursorUpdate > 200) {
        lastFirebaseCursorUpdate = now;
        update(dbRef, { pointer: { x: payload.pctX, y: payload.pctY, initiator: myRole } });
      }
    } else {
      const actionId = payload.actionId || ('click_' + Date.now() + '_' + Math.random().toString(36).substr(2, 5));
      update(dbRef, { last_action: { ...payload, actionId, initiator: myRole, timestamp: Date.now() } });
    }
  }
  if (event.data?.type === 'IFRAME_MOUSE_TRACK') {
    const now = Date.now();
    if (now - lastFirebaseCursorUpdate > 200) {
      lastFirebaseCursorUpdate = now;
      update(dbRef, { pointer: { x: event.data.pctX, y: event.data.pctY, initiator: myRole } });
    }
  }
});

function loadWorksheetFast(path) {
  if (!path) return;
  hideOverlay('waiting-overlay');
  showLoadingOverlay();
  const frame = document.getElementById('game-frame');
  const url = buildWorksheetUrl(path);
  frame.onload = () => { hideLoadingOverlay(); clearTimeout(window._loadFallback); };
  clearTimeout(window._loadFallback);
  window._loadFallback = setTimeout(() => hideLoadingOverlay(), 2000);
  if (frame.src === url || frame.getAttribute('src') === url) {
    hideLoadingOverlay();
    return;
  }
  frame.src = url;
}

function showLoadingOverlay() { showOverlay('loading-overlay'); }
function hideLoadingOverlay() { hideOverlay('loading-overlay'); }

function showToast(msg) {
  document.getElementById('toast-text').innerText = msg;
  const toast = document.getElementById('toast');
  toast.classList.remove('translate-y-24', 'opacity-0');
  setTimeout(() => toast.classList.add('translate-y-24', 'opacity-0'), 2500);
}
